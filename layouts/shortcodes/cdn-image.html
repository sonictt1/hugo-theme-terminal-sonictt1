{{/* cdn-image shortcode — renders an served from the site's CDN. Params: src (required) Path relative to the CDN root, e.g. "decks/nazgul.jpg" alt (optional) Alt text position (optional) CSS class for positioning: left (default), center, right style (optional) Inline style string width (optional) width attribute height (optional) height attribute CDN base URL is read from .Site.Params.cdn_base_url. Falls back to a root-relative path if cdn_base_url is not set. Usage: {{< cdn-image src="decks/nazgul.jpg" alt="Frodo pursued by Nazgul" position="center" >}} */}} {{ $base := .Site.Params.cdn_base_url | default "" }} {{ $base = strings.TrimRight "/" $base }} {{ $src := .Get "src" | strings.TrimLeft "/" }} {{ $url := printf "%s/%s" $base $src }}
